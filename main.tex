% ── 2. Operational problem ──────────────────────────────────────────────────
\begin{frame}{Conditioning works. What does it change?}
  \begin{columns}[c]
    \column{0.54\textwidth}
      \begin{itemize}\setlength\itemsep{0.6em}
        \item Conditioning is how every high-gradient device reaches its usable field.\src{a}
        \item Breakdown rate has an extreme field dependence, ${\sim}E^{30}$.\src{a}
        \item It tracks the number of \alert{high-field pulses}, not the number of breakdowns.\src{a,b}
        \item The field itself drives conditioning, not the arcs.\src{a,b}
      \end{itemize}
    \column{0.44\textwidth}

  \end{columns}
  \claim{Something in the metal stores the improvement. We do not know what.\src{a}}

% ── 3. MDDF model ───────────────────────────────────────────────────────────
\begin{frame}{Our model: breakdown as collective dislocation dynamics}
  \begin{columns}[c]
    \column{0.54\textwidth}
      \begin{itemize}\setlength\itemsep{0.5em}
        \item Field holding rises FCC$\to$BCC$\to$HCP\src{d} --- the order of barriers to dislocation motion.
        \item The \alert{MDDF} model: breakdown nucleates when the mobile-dislocation population, driven by Maxwell stress, runs away.\src{a,b}
        \item One model gets the ${\sim}E^{30}$ rate and its temperature dependence,\src{d} and was shown to reproduce the histogram of dark-current spikes.\src{c}
      \end{itemize}
    \column{0.44\textwidth}
      \centering
      \includegraphics[height=2.4cm,keepaspectratio]{mddf_critical_transition.png}\\[-1mm]
      {\scriptsize mobile-dislocation runaway}\\[0.02cm]
      \includegraphics[height=2.2cm,keepaspectratio]{mddf_rate_vs_field.png}\\[-1mm]
      {\scriptsize breakdown rate vs field}
  \end{columns}
  \claim{Central prediction: conditioning \emph{evolves the near-surface dislocation structure}.\src{d}}
  \sources{\srcl{a}Engelberg et al., PRL 120, 124801 (2018); \srcl{b}PRAB 22, 083501 (2019); \srcl{c}PRAB 23, 123501 (2020); \srcl{d}Wuensch et al., RMP 98, 025004 (2026).}
  \note{MDDF model is our own (2018--2020). Cartoon shows birth-death dynamics of the mobile-dislocation population: below critical density it decays, above it a runaway triggers nucleation. The plot reproduces the field dependence of breakdown rate. Key line for this talk: conditioning should leave an evolving dislocation structure just below the surface. Next slide: what that sentence means for a measurement.}
\end{frame}

% ── 4. Prior microscopy (2025 STEM) ─────────────────────────────────────────
\begin{frame}{STEM already saw it, at specific locations}
  \begin{columns}[c]

    \column{0.46\textwidth}
      \small
      \begin{itemize}\setlength\itemsep{0.05em}
        \item \alert{Direct} evidence of near-surface plasticity in conditioned regions.\src{a}
        \item Fewer dislocation walls at \SI{300}{K} and \SI{30}{K}; denuded layer clearest at \SI{30}{K}.\src{a}
        \item Fits the clue that soft Cu conditions far more slowly than hard Cu: the starting population matters.\src{c}
        \item FIB preparation plants its own dislocations; trust only large-scale structure.\src{b}
        \item One cross-section cannot speak for the whole electrode.
      \end{itemize}
  \end{columns}
  \claim{Seen at specific locations. The large-area view was missing.}
  \sources{\srcl{a}Jacewicz et al., JAP 137, 193302 (2025); \srcl{b}HUJI STEM (I.~Popov), MeVArc 2025; \srcl{c}Korsb\"ack et al., PRAB 23, 033102 (2020).}
  \note{2025 STEM images. Dark band is the protective coating; just beneath it the field-exposed region is nearly free of dislocation walls while the reference keeps walls continuous to the surface. Published wording is conservative: 300 K shows significant reduction of walls; the denuded zone is clearly visible at 30 K. The paper gives no thickness; 200 nanometers is read from images, the number the manuscript uses. Hard-versus-soft copper contrast (Korsb\"ack 2020) is an independent macroscopic clue: hard copper conditioned within about 25 million pulses; soft samples began saturating near 600 million or stayed unsaturated longer, consistent with different starting dislocation populations. Two lessons: FIB preparation generates dislocations and loops, so only larger-scale structure is trustworthy; TEM and STEM is intrinsically local. Last year I asked for more comparative forensics: other materials, more samples. That is exactly the gap next slides fill.}
\end{frame}
