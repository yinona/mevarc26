% ── 2. Operational problem ──────────────────────────────────────────────────
\begin{frame}{Conditioning works. What does it change?}
  \begin{columns}[c]
    \column{0.54\textwidth}
      \begin{itemize}\setlength\itemsep{0.6em}
        \item Conditioning is how every high-gradient device reaches its usable field.
        \item Breakdown rate has an extreme field dependence, ${\sim}E^{30}$.
        \item It tracks the number of \alert{high-field pulses}, not the number of breakdowns.
        \item The field itself drives conditioning, not the arcs.
      \end{itemize}
    \column{0.44\textwidth}
      \begin{tikzpicture}[x=1cm,y=1cm]
        \draw[->,very thick,ink] (0,0) -- (4.25,0) node[below] {exposure};
        \draw[->,very thick,ink] (0,0) -- (0,3.2) node[above] {field holding};
        \draw[very thick,teal] (0.2,.35) .. controls (1.1,.6) and (1.2,1.5) .. (2.1,1.65)
          .. controls (3.0,1.8) and (3.4,2.65) .. (4.5,2.82);
        \node[fill=white,draw=copper,rounded corners,align=center] at (3.05,.85)
          {which material\\variable is\\changing? {\color{copper}\bfseries ?}};
      \end{tikzpicture}
  \end{columns}
  \claim{Something in the metal stores the improvement. We do not know what.}
  \source{Wuensch et al., Rev. Mod. Phys. 98, 025004 (2026); Degiovanni et al., PRAB (2016). See also W.~Wuensch, this morning (09:15).}
  \note{Two macroscopic clues constrain the answer: the near-thirtieth-power field dependence, and conditioning following pulse count not breakdown count. Both say the applied field, not the arc, drives the work. Look for a field-driven change in the metal itself.}
\end{frame}

% ── 3. MDDF model ───────────────────────────────────────────────────────────
\begin{frame}{Our model: conditioning as collective dislocation dynamics}
  \begin{columns}[c]
    \column{0.54\textwidth}
      \begin{itemize}\setlength\itemsep{0.5em}
        \item Field holding rises FCC$\to$BCC$\to$HCP --- the order of barriers to dislocation motion.
        \item The \alert{MDDF} model: breakdown nucleates when the mobile-dislocation population, driven by Maxwell stress, runs away.
        \item One model gets the ${\sim}E^{30}$ rate, its temperature dependence, and the dark-current spikes.
      \end{itemize}
    \column{0.44\textwidth}
      \centering
      \includegraphics[height=2.85cm,keepaspectratio]{mddf_critical_transition.png}\\[0.1cm]
      \includegraphics[height=2.6cm,keepaspectratio]{mddf_rate_vs_field.png}
  \end{columns}
  \claim{Central prediction: conditioning \emph{evolves the near-surface dislocation structure}.}
  \source{Engelberg et al., PRL 120, 124801 (2018); PRAB 22, 083501 (2019); PRAB 23, 123501 (2020).}
  \note{MDDF model is our own (2018--2020). Cartoon shows birth-death dynamics of the mobile-dislocation population: below critical density it decays, above it a runaway triggers nucleation. The plot reproduces the field dependence of breakdown rate. Key line for this talk: conditioning should leave an evolving dislocation structure just below the surface. Everything after this slide tests that sentence.}
\end{frame}

% ── 4. Prior microscopy (2025 STEM) ─────────────────────────────────────────
\begin{frame}{STEM already saw it, at single spots}
  \begin{columns}[c]
    \column{0.52\textwidth}
      \centering
      \includegraphics[width=\linewidth,height=4.1cm,keepaspectratio]{stem_denuded_zone.png}\\[-1mm]
      {\scriptsize Cross-sectional STEM of conditioned hard (as-machined) Cu: dislocation-poor zone in the top ${\sim}\SI{200}{nm}$.}
    \column{0.46\textwidth}
      \small
      \begin{itemize}\setlength\itemsep{0.05em}
        \item \alert{Direct} evidence of near-surface plasticity in conditioned regions.
        \item Fewer dislocation walls at \SI{300}{K} and \SI{30}{K}; denuded layer clearest at \SI{30}{K}.
        \item Fits the clue that soft Cu conditions far more slowly than hard Cu: the starting population matters.
        \item FIB preparation plants its own dislocations; trust only large-scale structure.
        \item One cross-section cannot speak for the whole electrode.
      \end{itemize}
  \end{columns}
  \claim{Seen at one spot. The large-area comparison was still missing.}
  \source{Jacewicz et al., JAP 137, 193302 (2025); HUJI STEM (I.~Popov), MeVArc 2025; Korsb\"ack et al., PRAB 23, 033102 (2020).}
  \note{2025 STEM images. Dark band is the protective coating; just beneath it the field-exposed region is nearly free of dislocation walls while the reference keeps walls continuous to the surface. Published wording is conservative: 300 K shows significant reduction of walls; the denuded zone is clearly visible at 30 K. The paper gives no thickness; 200 nanometers is read from images, the number the manuscript uses. Hard-versus-soft copper contrast (Korsb\"ack 2020) is an independent macroscopic clue: hard copper conditioned within about 25 million pulses; soft samples began saturating near 600 million or stayed unsaturated longer, consistent with different starting dislocation populations. Two lessons: FIB preparation generates dislocations and loops, so only larger-scale structure is trustworthy; TEM and STEM is intrinsically local. Last year I asked for more comparative forensics: other materials, more samples. That is exactly the gap next slides fill.}
\end{frame}

% ── 5. Maxwell stress + VHCF, and the bridge to results ──────────────────────
\begin{frame}{A tiny stress, repeated a billion times}
  \begin{columns}[c]
    \column{0.42\textwidth}
      \begin{beamercolorbox}[rounded=true,sep=.35cm]{block body}
        \centering
        \textbf{Maxwell stress}\\[0.15cm]
        $\displaystyle \sigma_M=\frac{\varepsilon_0 E^2}{2}$\\[0.28cm]
        at $\SI{80}{MV/m}$: \alert{$\SI{0.028}{MPa}$}\\[0.2cm]
        \scriptsize ${\sim}10^3\times$ below the yield of annealed Cu
      \end{beamercolorbox}
      \vspace{0.2cm}
      \centering
      \begin{tikzpicture}[x=1cm,y=1cm,font=\scriptsize]
        \draw[->,ink] (0,0) -- (5.0,0) node[below left=-1pt,font=\tiny] {$t$};
        \draw[thick,copper] (0,0) -- (0.4,0) -- (0.4,0.55) -- (0.5,0.55) -- (0.5,0) -- (1.4,0) -- (1.4,0.55) -- (1.5,0.55) -- (1.5,0) -- (2.4,0) -- (2.4,0.55) -- (2.5,0.55) -- (2.5,0) -- (3.0,0);
        \node[color=copper] at (3.6,0.3) {$\cdots\times10^{9}$};
        \draw[<->,muted] (0.5,0.68) -- (1.4,0.68) node[midway,above=-1pt,font=\tiny] {$\SI{1}{ms}$};
        \node[color=muted,font=\tiny,anchor=south] at (0.45,0.57) {$\SI{1}{\micro s}$};
        \node[anchor=west,color=muted,font=\tiny] at (0,-0.35) {$\sigma_M(t)$: the only cyclic load};
      \end{tikzpicture}
    \column{0.55\textwidth}
      \begin{itemize}\setlength\itemsep{0.5em}
        \item Fatigue at $10^{8}$--$10^{10}$ cycles rearranges dislocations irreversibly, well below yield.
        \item Here: ${\sim}10^{9}$ conditioning pulses.
        \item Pulsed DC means \alert{no RF pulsed heating} and no bulk currents. Maxwell stress is the only cyclic load.
        \item MDDF is \alert{threshold-free}: no minimum stress amplitude.
      \end{itemize}
  \end{columns}
  \claim{The model predicts a large-area signature. Last year we asked for it; this year we looked.}
  \source{Mughrabi, MMTB 40, 431 (2009); Stanzl-Tschegg et al., Int.\ J.\ Fatigue 29, 2050 (2007). Field-driven deformation at the nanoscale: J.~Wang (Meng et al.) and V.~Zadin, this afternoon (16:00, 16:30).}
  \note{Maxwell stress is three orders of magnitude below yield. The point is not amplitude but accumulation. VHCF shows oscillatory loading well below yield still rearranges dislocations irreversibly for a billion cycles---our regime. Pulsed-DC is cleaner than RF structures: no RF magnetic field, no pulsed surface heating, no bulk Ohmic current. Maxwell stress is the only cyclic mechanical load. Earlier RF-based studies could never fully separate thermal fatigue from field effects. MDDF is threshold-free: does not need the stress to exceed any yield-like value. That argument bridges the 2025 model-and-microscopy story to this year's large-area measurement.}
\end{frame}

All ROIs deliberately far from craters, measuring cumulative material response to the field, not arc damage. % ── 10. Three-tier hierarchy ─────────────────────────────────────────────────
\begin{frame}{The signal separates into three exposure tiers}
  \begin{columns}[c]

  \source{Ashkenazy et al., \condref.}
  \note{Three tiers: high-field center and edge highest and similar; field-exposed periphery intermediate; unexposed references lowest. Intermediate value consistent with residual fringe-field exposure. The 75 percent figure characterizes this cathode and history, not a universal constant.}
\end{frame}
