% ── 2. Operational problem ──────────────────────────────────────────────────
\begin{frame}{Conditioning works. What does it change?}
  \begin{columns}[c]
    \column{0.54\textwidth}
      \begin{itemize}\setlength\itemsep{0.6em}
        \item Conditioning is how every high-gradient device reaches its usable field.
        \item Breakdown rate has an extreme field dependence, ${\sim}E^{30}$.
        \item It tracks the number of \alert{high-field pulses}, not the number of breakdowns.
        \item The field itself drives conditioning, not the arcs.
      \end{itemize}
    \column{0.44\textwidth}
      \begin{tikzpicture}[x=1cm,y=1cm]
        \draw[->,very thick,ink] (0,0) -- (4.25,0) node[below] {exposure};
        \draw[->,very thick,ink] (0,0) -- (0,3.2) node[above] {field holding};
        \draw[very thick,teal] (0.2,.35) .. controls (1.1,.6) and (1.2,1.5) .. (2.1,1.65)
          .. controls (3.0,1.8) and (3.4,2.65) .. (4.5,2.82);
        \node[fill=white,draw=copper,rounded corners,align=center] at (3.05,.85)
          {which material\\variable is\\changing? {\color{copper}\bfseries ?}};
      \end{tikzpicture}
  \end{columns}
  \claim{Something in the metal stores the improvement. We do not know what.}
  \source{Wuensch et al., Rep. Prog. Phys. (2026); Degiovanni et al., PRAB (2016). See also W.~Wuensch, this morning (09:15).}
  \note{Open on the operational fact and the puzzle. Do not introduce E\_S here. Two macroscopic clues already constrain the answer: the near-thirtieth-power field dependence, and that conditioning follows pulse count rather than breakdown count. Both say the applied field, not the arc, is doing the work---so we should look for a field-driven change in the metal itself.}
\end{frame}

% ── 3. MDDF model ───────────────────────────────────────────────────────────
\begin{frame}{Our model: conditioning as collective dislocation dynamics}
  \begin{columns}[c]
    \column{0.54\textwidth}
      \begin{itemize}\setlength\itemsep{0.5em}
        \item Field holding rises FCC$\to$BCC$\to$HCP --- the order of barriers to dislocation motion.
        \item The \alert{MDDF} model: breakdown nucleates when the mobile-dislocation population, driven by Maxwell stress, runs away.
        \item One model gets the ${\sim}E^{30}$ rate, its temperature dependence, and the dark-current spikes.
      \end{itemize}
    \column{0.44\textwidth}
      \centering
      \includegraphics[height=2.85cm,keepaspectratio]{mddf_critical_transition.png}\\[0.1cm]
      \includegraphics[height=2.6cm,keepaspectratio]{mddf_rate_vs_field.png}
  \end{columns}
  \claim{Central prediction: conditioning \emph{evolves the near-surface dislocation structure}.}
  \source{Engelberg et al., PRL 120, 124801 (2018); PRAB 22, 083501 (2019); PRAB 23, 123501 (2020).}
  \note{The MDDF model is our own (2018--2020). The cartoon shows the birth--death dynamics of the mobile-dislocation population: below a critical density the population decays, above it a runaway triggers nucleation. The plot is the reproduced field dependence of the breakdown rate. The key line for this talk is the prediction highlighted at the bottom: conditioning should leave an evolving dislocation structure just below the surface. Everything after this slide is a test of that sentence.}
\end{frame}

% ── 4. Prior microscopy (2025 STEM) ─────────────────────────────────────────
\begin{frame}{STEM already saw it, at single spots}
  \begin{columns}[c]
    \column{0.52\textwidth}
      \centering
      \includegraphics[width=\linewidth,height=4.1cm,keepaspectratio]{stem_denuded_zone.png}\\[-1mm]
      {\scriptsize Cross-sectional STEM of conditioned hard (as-machined) Cu: dislocation-denuded zone in the top ${\sim}\SI{200}{nm}$.}
    \column{0.46\textwidth}
      \small
      \begin{itemize}\setlength\itemsep{0.2em}
        \item First \alert{direct} structural evidence of field-driven near-surface plasticity.
        \item Seen at both \SI{300}{K} and \SI{30}{K}.
        \item Fits the clue that soft Cu conditions far more slowly than hard Cu: the starting population matters.
        \item FIB preparation plants its own dislocations; trust only large-scale structure.
        \item One cross-section cannot speak for the whole electrode.
      \end{itemize}
  \end{columns}
  \claim{Confirmed at one spot. The large-area comparison was still missing.}
  \source{Jacewicz et al., J.\ Appl.\ Phys.\ 137, 193302 (2025) --- Uppsala/FREIA; Popov, HUJI. Korsb\"ack et al., PRAB 23, 033102 (2020). Shown at MeVArc 2025.}
  \note{These are our 2025 STEM images. The dark band is the protective coating over the surface; just beneath it the field-exposed region is denuded of dislocations, while the reference stays dislocation-rich to the surface. The hard-versus-soft copper contrast (Korsb\"ack 2020) is an independent macroscopic clue pointing the same way: about 25 million pulses to condition hard copper versus roughly 600 million for soft, consistent with different starting dislocation populations. Two lessons carried forward: FIB sample preparation itself generates dislocations and loops---real red herrings---so only larger-scale structure is trustworthy; and TEM/STEM is intrinsically local. Last year I closed by asking for more comparative forensics: other materials, more samples. That is exactly the gap the next slides fill.}
\end{frame}

% ── 5. Maxwell stress + VHCF, and the bridge to results ──────────────────────
\begin{frame}{A tiny stress, repeated a billion times}
  \begin{columns}[c]
    \column{0.42\textwidth}
      \begin{beamercolorbox}[rounded=true,sep=.35cm]{block body}
        \centering
        \textbf{Maxwell stress}\\[0.15cm]
        $\displaystyle \sigma_M=\frac{\varepsilon_0 E^2}{2}$\\[0.28cm]
        at $\SI{80}{MV/m}$: \alert{$\SI{0.028}{MPa}$}\\[0.2cm]
        \scriptsize ${\sim}10^3\times$ below the yield of annealed Cu
      \end{beamercolorbox}
      \vspace{0.2cm}
      \centering
      \begin{tikzpicture}[x=1cm,y=1cm,font=\scriptsize]
        \draw[->,ink] (0,0) -- (5.0,0) node[below left=-1pt,font=\tiny] {$t$};
        \draw[thick,copper] (0,0) -- (0.4,0) -- (0.4,0.55) -- (0.5,0.55) -- (0.5,0) -- (1.4,0) -- (1.4,0.55) -- (1.5,0.55) -- (1.5,0) -- (2.4,0) -- (2.4,0.55) -- (2.5,0.55) -- (2.5,0) -- (3.0,0);
        \node[color=copper] at (3.6,0.3) {$\cdots\times10^{9}$};
        \draw[<->,muted] (0.5,0.68) -- (1.4,0.68) node[midway,above=-1pt,font=\tiny] {$\SI{1}{ms}$};
        \node[color=muted,font=\tiny,anchor=south] at (0.45,0.57) {$\SI{1}{\micro s}$};
        \node[anchor=west,color=muted,font=\tiny] at (0,-0.35) {$\sigma_M(t)$: the only cyclic load};
      \end{tikzpicture}
    \column{0.55\textwidth}
      \begin{itemize}\setlength\itemsep{0.5em}
        \item Fatigue at $10^{8}$--$10^{10}$ cycles rearranges dislocations irreversibly, well below yield.
        \item Here: ${\sim}10^{9}$ conditioning pulses.
        \item Pulsed DC means \alert{no RF pulsed heating} and no bulk currents. Maxwell stress is the only cyclic load.
        \item MDDF is \alert{threshold-free}: no minimum stress amplitude.
      \end{itemize}
  \end{columns}
  \claim{The model predicts a large-area signature. Last year we asked for it; this year we looked.}
  \source{Mughrabi, MMTB 40, 431 (2009); Stanzl-Tschegg et al., Int.\ J.\ Fatigue 29, 2050 (2007). Field-driven deformation at the nanoscale: G.~Meng and V.~Zadin, this afternoon (16:00, 16:30).}
  \note{Head off the yield-stress objection: the Maxwell stress is three orders of magnitude below yield. The point is not amplitude but accumulation. VHCF shows that oscillatory loading well below yield still rearranges dislocations irreversibly when sustained for a billion cycles, which is our regime. The pulsed-DC system is also a cleaner experiment than RF structures: there is no RF magnetic field, no pulsed surface heating, and no bulk Ohmic current, so the Maxwell stress is the only cyclic mechanical load---earlier RF-based studies could never fully separate thermal fatigue from field effects. And the MDDF description is threshold-free, so it does not need the stress to exceed any yield-like value. That argument is the bridge from the 2025 model-and-microscopy story to this year's large-area measurement.}
\end{frame}

% ── 7. EBSD away from craters ────────────────────────────────────────────────
\begin{frame}{EBSD maps plastic activity far from breakdown craters}
  \begin{columns}[c]
    \column{0.47\textwidth}
      \includegraphics[width=\linewidth]{sem_lowmag.png}
      \vspace{-0.15cm}
      {\scriptsize Cleaned ROI is hundreds of microns from visible craters.}
    \column{0.49\textwidth}
      \begin{itemize}\setlength\itemsep{0.45em}
        \item Large area where TEM could only sample a point
        \item $\SI{500}{\micro m}$-wide maps; $\SI{3}{\micro m}$ step (lateral only); information depth a few tens of \si{nm}
        \item ${\sim}24{,}000$ indexed points per ROI
        \item LAM, KAM, and LOS probe intragrain orientation gradients
        \item Misorientation is a proxy for GND density
      \end{itemize}
  \end{columns}
  \claim{We measure the cumulative response to the field, not crater damage.}
  \source{Ashkenazy et al., \condref; information depth after Chen et al., Ultramicroscopy 111, 1488 (2011).}
  \note{EBSD is chosen precisely because it is large-area where TEM is local. A shallow FIB clean removes oxide, under 10 nm by AFM, then we map orientation gradients. The 3 micron step and kernel fix only the lateral resolution; the backscattered electrons that carry the pattern come from the top few tens of nanometers of copper at 15 kV, so this is a near-surface measurement. All ROIs are deliberately far from craters, so we measure the cumulative material response to the field, not arc damage. LAM is the primary metric; KAM and LOS check that the result does not depend on one analysis choice.}
\end{frame}

\begin{frame}{The signal separates into three exposure tiers}

% ── 11. Distribution tail ────────────────────────────────────────────────────
\begin{frame}{Field exposure builds a high-misorientation tail}
  \begin{columns}[c]
    \column{0.60\textwidth}
      \figorbox{low_angle_overlay.pdf}{width=\linewidth,height=4.9cm,keepaspectratio}{low-angle LAM distributions with gamma fits: field-exposed curves shift right and broaden}
    \column{0.36\textwidth}
      \begin{beamercolorbox}[rounded=true,sep=.28cm]{block body}
        \centering
        \textbf{LAM tail above $\ang{2}$}\\[0.2cm]
        \alert{${\sim}0.14$} field-exposed\\
        ${\sim}0.016$ reference\\[0.15cm]
        \textbf{${\sim}8\times$ increase}
      \end{beamercolorbox}
      \vspace{0.3cm}
      {\small Gamma shape $k\approx2.7$ fixed; the scale roughly doubles.}
  \end{columns}
  \claim{The whole distribution stretches; a uniform offset cannot do that.}
  \source{Ashkenazy et al., \condref.}
  \note{The mean is not the whole story. The low-angle distribution broadens and grows a long tail: the fraction above two degrees rises roughly eightfold. The gamma shape parameter stays near 2.7 while the scale roughly doubles---a fixed-shape stretch, the signature of progressive deformation. LAM, KAM, and LOS agree, and KS, chi-squared, and Anderson--Darling all separate the distributions; every center or edge ROI differs from the unexposed ones with a KS distance above 0.25. I quote effect sizes and grain-corrected errors rather than the extreme pixel-pair p-values.}
\end{frame}

   screened interior field $\leftrightarrow$ screened internal stress of the depleted layer\\[0.05em]

Note the contrast on the right: the electric field itself is screened by surface charge within a nanometre and never enters the metal, whereas the load enters as elastic stress, so the electrostatic screening length says nothing about depth. The STEM denuded zone in hard copper is about eight of these screening lengths thick; no theory yet treats a cyclically loaded free surface, so that thickness is an observation, not a prediction.}
\end{frame}
