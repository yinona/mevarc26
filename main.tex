% ── 3. MDDF model ───────────────────────────────────────────────────────────
\begin{frame}{Our model: breakdown as collective dislocation dynamics}
  \begin{columns}[c]
    \column{0.54\textwidth}
      \begin{itemize}\setlength\itemsep{0.5em}
        \item Field holding rises FCC$\to$BCC$\to$HCP\src{d} --- matching the order of barriers to dislocation motion.
        \item The \alert{MDDF} model: breakdown nucleates when the mobile-dislocation population, driven by Maxwell stress, runs away.\src{a,b}
        \item The same model, fitted to the ${\sim}E^{30}$ rate and its temperature dependence,\src{d} then reproduced the histogram of dark-current spikes.\src{c}
      \end{itemize}
    \column{0.44\textwidth}
      \centering
      \includegraphics[height=2.4cm,keepaspectratio]{mddf_critical_transition.png}\\[-1mm]
      {\scriptsize mobile-dislocation runaway}\\[0.02cm]
      \includegraphics[height=2.2cm,keepaspectratio]{mddf_rate_vs_field.png}\\[-1mm]
      {\scriptsize breakdown rate vs field}
  \end{columns}
  \claim{Central prediction: conditioning \emph{evolves the near-surface dislocation structure}.\src{d}}
  \sources{\srcl{a}Engelberg et al., PRL 120, 124801 (2018); \srcl{b}PRAB 22, 083501 (2019); \srcl{c}PRAB 23, 123501 (2020); \srcl{d}Wuensch et al., RMP 98, 025004 (2026).}
  \note{Our model treats breakdown as collective dislocation dynamics. Field holding rises from FCC to BCC to HCP metals, matching the order of barriers to dislocation motion. In MDDF, the mobile dislocations driven by the Maxwell stress fluctuate, and breakdown nucleates when that population runs away. We fitted the same model to the thirtieth-power rate and its temperature dependence, and it then reproduced the histogram of dark-current spikes, which was a prediction. Its central prediction for this talk is that conditioning evolves the near-surface dislocation structure. Next: what that means for a microscope.}
\end{frame}
